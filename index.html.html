models/vehicle_service.py Python 3 + Odoo ORM 
from odoo import models, fields, api

class VehicleServiceRequest(models.Model):
    _name = 'vehicle.service.request'
    _description = 'Vehicle Repair and Maintenance Request'

    name = fields.Char(string='Service #', required=True)
    partner_id = fields.Many2one('res.partner', string='Customer', required=True)
    vehicle_plate = fields.Char(string='License Plate', required=True)
    total_cost = fields.Float(string='Total (INR)', compute='_compute_total')

    @api.depends('line_ids.price_subtotal')
    def _compute_total(self):
        for record in self:
            record.total_cost = sum(record.line_ids.mapped('price_subtotal'))
